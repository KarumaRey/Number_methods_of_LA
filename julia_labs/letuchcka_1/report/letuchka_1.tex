\section{Практическая реализация}
\par

Функция progonka() решает систему линейных алгебраических уравнений с трехдиагональной матрицей. Она принимает четыре вектора a, b, c и f как входные параметры, где a, b, c представлюят элементы трёх диагоналей соответственно, а f - результирующий вектор.
\par
Внутри функции progonka() создается два вектора p и q длиной n, которые используются для хранения промежуточных результатов.
\par
Функция norm() вычисляет евклидову норму вектора x, полученного методомм прогонки, а функция err() вычисляет погрешность.

\par
Код программы представлен ниже:
\begin{lstlisting}
n = 1000

function progonka(a, b, c, d)

    p = zeros(n)
    q = zeros(n)

    p[1] = c[1] / b[1]
    q[1] = f[1] / b[1]
    for i in 2:n-1
        k = (b[i] - p[i-1]*a[i])
        p[i] = c[i] / k
        q[i] = (f[i] - q[i-1]*a[i]) / k
    end
    q[n] = (f[n] - q[n-1]*a[n]) / (b[n] - p[n-1]*a[n])

    x = zeros(n)
    x[n] = q[n]
    for i in n-1:-1:1
        x[i] = q[i] - p[i]*x[i+1]
    end
    
    return x
end

function norm(x)
    sum = 0.0
    for element in x
        sum += element^2
    end
    return sqrt(sum)
end

function err(a, b, c, f, x)
    f1 = zeros(n)
    f1[1] = b[1] * x[1] + c[1] * x[2]
    f1[n] = a[n] * x[n-1] + b[n] * x[n]
    for i in 2:n-1
        f1[i] = a[i] * x[i-1] + b[i] * x[i] + c[i] * x[i+1]
    end
    
    for i in 1:n
       # print(f1[i], " ")
    end
    println()
    
    return norm(f1 - f)
end

a = zeros(n)
b = zeros(n)
c = zeros(n)
a[n] = 1.0
c[1] = 1.0
b[1] = 4.0
b[n] = 4.0
for i in 2:n-1
    a[i] = 1.0
    b[i] = 4.0
    c[i] = 1.0
end

f = zeros(n)

    f[i] = 6.0
