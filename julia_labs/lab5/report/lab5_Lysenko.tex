\section{Цель работы}\label{Sect::task}
\par
\renewcommand{\labelenumii}{\arabic{enumi}.\arabic{enumii}.}
Реализовать метод вычисления собственных значений и собственных векторов симметричной матрицы методом А.М. Крылова.

\section{Задачи}\label{Sect::task}
\par
\renewcommand{\labelenumii}{\arabic{enumi}.\arabic{enumii}.}
\begin{enumerate} 
    \item Реализовать метод вычисления собственных значений и собственных векторов симметричной матрицы методом А.М. Крылова.
    \item Реализовать метод поиска собственных значений действительной симметричной матрицы A размером 4х4.
    \item Проверить решение на матрице приведенной в презентации.
    \item Продемонстрировать работу приложения для произвольных симметричных матриц размером n х n с учетом выполнения пунктов приведенных выше.
    \item Сравнить результаты с полученными методом Данилевского.
\end{enumerate}

\pagebreak
\section{Практическая реализация}
\par
Сравнение результатов, полученных методами Данилевского и Крылова, реализовано по следущему принципу:
\par
При сравнении собственных значений вычисляются их разницы с собственными значениями, полученными библиотечным методом. После берётся самая большая разница для каждого из методов.
\par
При сравнении собственных векторов вычисляются углы между векторами, полученными каждым из методов, с векторами, вычесленными библиотечным методом. После берётся самый большой угол для каждого из методов. Углы вычеслены в радианнах.
\par

using Random

eps = 1e-10

function generate_symmetric_matrix(n)
    A = rand(Float64, n, n)
    for i in 1:n
        for j in i+1:n
            A[i,j] = A[j, i]
        end
    end  
    return A
end

function roots_for_Daniel(coefficients)
    x = -reverse(coefficients)
    append!(x, 1)
    p = Polynomial(x)
    
    return roots(p)
end


function danilevsky_method(A)
    n = size(A, 1)
    lambda_values = Vector{Float64}()
    lambda_vectors = Vector{Vector{Float64}}()
    main_B = I(n)

    B = A

    for k in 1:n-1
        B = zeros(Float64, n, n)
        for i in 1:n
            B[i,i] = 1.0
        end
        for j in 1:n
            B[n-k,j] = - A[n+1-k,j] / A[n+1-k, n-k]
            if j==n-k
                B[n-k,j] = 1 / A[n+1-k, n-k]
            end
        end

        A = inv(B) * A * B

        main_B *= B        
    end

    lambda_values = roots_for_Daniel(A[1, :])
    
    for val in lambda_values
        my_vector = Vector{Float64}()
        for i in 0:n-1
            push!(my_vector, val^i)
        end
        
        my_vector = main_B * reverse(my_vector)
        
        push!(lambda_vectors, my_vector / norm(my_vector))
    end
    return lambda_values, lambda_vectors
end

function roots_for_Krylov(coefficients)
    x = -coefficients
    append!(x, 1)
    p = Polynomial(x)
    
    return roots(p)
end


function krylov_method(A)
    n = size(A, 1)
    c = zeros(Float64, n)
    c[1] = 1.0
    c_vectors = Vector{Vector{Float64}}()
    push!(c_vectors, c)
    lambda_vectors = Vector{Vector{Float64}}()

    for i in 1:n
        c = A * c
        push!(c_vectors, c)
    end

    с_matrix = zeros(Float64, n, n)

    for i in 1:n
        for j in 1:n
            с_matrix[i,j] = c_vectors[j][i] 
        end
    end

    
    Alu = qr(с_matrix)
    p = Alu \ c_vectors[n+1]

    lambda_values = roots_for_Krylov(p)

    p = reverse(p)
    for val in lambda_values
        
        betas = Vector{Float64}()
        push!(betas, 1)
        for i in 2:n
            push!(betas, val*betas[i-1] - p[i-1]) 
            
        end

        
        x = reverse(с_matrix) * betas
        push!(lambda_vectors, reverse(x / norm(x) * sign(x[1])))
        
    end

    return lambda_values, lambda_vectors

end

function check_viete(lambda_values, A)
    trace_A = tr(A)
    det_A = det(A)
    sum_lambda_values = sum(lambda_values)
    product_lambda_values = prod(lambda_values)

    if isapprox(sum_lambda_values, trace_A) && isapprox(product_lambda_values, det_A)
        println("Проверка теоремы Виета пройдена")
    else
        println("Проверка теоремы Виета НЕ пройдена")
    end
end

function check_gershgorin(lambda_values, A)
    n = size(A, 1)
    is_valid = true

    for i in 1:n
        row_sum = sum(abs.(A[i, :])) - abs(A[i, i])
        if abs(lambda_values[i] - A[i, i]) > row_sum
            is_valid = false
        end
    end

    if is_valid
        println("Проверка кругов Гершгорина пройдена")
    else
        println("Проверка кругов Гершгорина НЕ пройдена")
    end
end


function are_orthogonal(vectors)
    n = size(vectors, 1)

    is_ortagonal = true
    for i in 1:n
        for j in 1:n
            if i != j
                dot_product = dot(vectors[i], vectors[j])
                if eps > dot_product
                    dot_product = 0
                end
                is_ortagonal = isapprox(dot_product, 0)
            end
            if !is_ortagonal
                break
            end
        end
        if !is_ortagonal
            break   
        end
    end

    if is_ortagonal
        println("Собственные вектора ортогональны")
    else
        println("Собственные вектора не ортогональны")
    end
end

function compare_values(lambda_values_danilevsky, lambda_values_krylov)
    diff = abs.(lambda_values_danilevsky - lambda_values_krylov)
    return maximum(diff)
end

function compare_vectors(lambda_vectors_danilevsky, lambda_vectors_krylov, n)
    angels = Vector{Float64}()
    for i in 1:n
        push!(angels, dot(lambda_vectors_danilevsky[i], lambda_vectors_krylov[i]) / (norm(lambda_vectors_danilevsky[i]) * norm(lambda_vectors_krylov[i]))
    end
    return maximum(angels)
end
