using LinearAlgebra


function Mysvd(A)
    m, n = size(A)
    
    mat = A' * A
    
    eigvals, V = eigen(mat)
    
    idx = sortperm(eigvals, rev=true)
    eigvals = eigvals[idx]
    V = V[:, idx]
    
    S = sqrt.(eigvals)
    
    lambda = zeros(m, n)
    for i in 1:min(m, n)
        lambda[i, i] = S[i]
    end
    
    U = zeros(m, m)
    for i in 1:size(V, 2)
        U[:, i] = A * V[:, i] / S[i]
    end
    
    return U, lambda, V
end

function calculate_error(A, A_approx)
    return norm(A - A_approx)
end


n = 5
A = rand(n, n) * 100
println("Матрица A:")
for row in eachrow(A)
    for x in row
        print(x, "  ")
    end
    println()
end
println()

U, lambda, V = Mysvd(A)

println("Матрица U:")
for row in eachrow(U)
    for x in row
        print(x, "  ")
    end
    println()
end
println()




println("Матрица lambda:")
for row in eachrow(lambda)
    for x in row
        print(x, "  ")
    end
    println()
end
println()

println("Матрица V:")
for row in eachrow(V)
    for x in row
        print(x, "  ")
    end
    println()
end
println()



A_approx = U * lambda * V'
println("Востановленная матрица A:")
for row in eachrow(A_approx)
    for x in row
        print(x, "  ")
    end
    println()
end
println()

error = calculate_error(A, A_approx)
println("Погрешность разложения:")
println(error)
\end{lstlisting}
