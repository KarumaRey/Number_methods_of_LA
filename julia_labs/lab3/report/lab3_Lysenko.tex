\section{Задачи}\label{Sect::task}
\par
\renewcommand{\labelenumii}{\arabic{enumi}.\arabic{enumii}.}
\begin{enumerate} 
    \item Реализовать автоматическое вычисление оценки относительной ошибки округления и ошибки в передаваемых данных для больших объёмов данных.
    \item Реализовать автоматическое вычисление оценки относительной ошибки округления и ошибки в передаваемых данных для значений из индивидуального варианта. 
    \item Сравнение вычисленных данных с полученными на бумаге.
\end{enumerate}

\pagebreak
\section{Практическая реализация}
\par
Для получения ошибки округления необходима функция, с погрешностью вычисляющая вектор х в выражении Ax = f. В качестве таковой была взята функция из предыдущей лабораторной работы, реализующая классический метод Гаусса. Данный метод был переопределён таким образом, чтобы возвращать помимо вектора х ещё и g(A) (коэффициент роста).
\par
При вычислении ошибки передаваемых данных для получения вектора х использовались уже имеющиеся возможности языка. 
\par
При выдаче задания было сказано находить норму по формуле Евклида, однако в письменной работе использовалась формула равномерной нормы, поэтому в данной лабораторной работе были реализованы методы для нахождения обеих норм ( norm() и matrix-norm() - для поиска норм вектора и матрицы по Евклиду, inf-norm() и inf-matrix-norm() - для поиска равномерных норм вектора и матрицы).
\par

Ниже представлен код программы:

n = 2

function gauss_classic(A, F)
    max = maximum(abs.(A))
    max_elems = zeros(Float64, n-1)

    for k in 1:n-1
        for i in k+1:n
            factor = A[i, k] / A[k, k]
            A[i, k:end] .-= factor * A[k, k:end]
            F[i] -= factor * F[k]
        end

        max_elems[k] = maximum(abs.(A)) / max     
    end

    g_a = maximum(max_elems)

    X = zeros(Float64, n)
    for i in n:-1:1
        X[i] = F[i] / A[i, i]
        for j in i-1:-1:1
            F[j] -= A[j, i] * X[i]
        end
    end

    return g_a, X
end


function norm(x)
    sum = 0.0
    for element in x
        sum += element^2
    end
    return sqrt(sum)
end

function matrix_norm(A)
    sum = 0.0
    for i in 1:n
        for j in 1:n
            #print(A[i, j], " ")
            sum += A[i, j] ^ 2
        end
        #println()
    end
    return sqrt(sum)
end

function inf_norm(x)
    return maximum(abs.(x))
end

function inf_matrix_norm(A)
    return maximum([sum(abs.(A[i, :])) for i in 1:size(A, 1)])
end
