\section{Цель работы}\label{Sect::task}
\par
\renewcommand{\labelenumii}{\arabic{enumi}.\arabic{enumii}.}
Реализовать метод А.А. Карацубы для перемножения длинных чисел.

\section{Задачи}\label{Sect::task}
\par
\renewcommand{\labelenumii}{\arabic{enumi}.\arabic{enumii}.}
\begin{enumerate} 
    \item Сгенерировать пары чисел размерности $2^{n}$, где n меняется от 1 до 15 (условно).
    \item Перемножить числа обычным методом и методом Карацубы.
    \item Сравнить время работы алгоритмов.
\end{enumerate}

Код программы:

using Random
using Plots

function random_number_array(n)
    return rand(0:9, n)
end


function standard_multiplication(x, y)
    n = length(x)
    result = zeros(Int, 2n)
    for i in 1:n
        carry = 0
        for j in 1:n
            result[i + j - 1] += x[i] * y[j] + carry
            carry = div(result[i + j - 1], 10)
            result[i + j - 1] = mod(result[i + j - 1], 10)
        end
        result[i + n] += carry
    end
    return result
end


function karatsuba(x, y)
    n = length(x)

    if n < 4
        return standard_multiplication(x, y)
    end

    m = div(n, 2)

    x1, x0 = x[1:m], x[m+1:end]
    y1, y0 = y[1:m], y[m+1:end]

    z2 = karatsuba(x1, y1)
    z0 = karatsuba(x0, y0)

    x1x0 = [x1[i] + x0[i] for i in 1:m]
    y1y0 = [y1[i] + y0[i] for i in 1:m]

    z1 = karatsuba(x1x0, y1y0)

    for i in 1:length(z2)
        z1[i] -= z2[i] + z0[i]
    end

    result = zeros(Int, 2n)

    for i in 1:length(z2)
        result[i] += z2[i]
        result[i + m] += z1[i]
        result[i + 2m] += z0[i]
    end

    return result
end

sizes = [2^i for i in 1:15]
std_times = Float64[]
karatsuba_times = Float64[]

for n in sizes
    x_arr = random_number_array(n)
    y_arr = random_number_array(n)

    std_time = @elapsed standard_multiplication(x_arr, y_arr)
    push!(std_times, std_time)

    karatsuba_time = @elapsed karatsuba(x_arr, y_arr)
    push!(karatsuba_times, karatsuba_time)
end

plot(sizes, std_times, label="Обычное умножение", xlabel="Количество разрядов (2^n)", ylabel="Время, с", legend=:topright, title="Сравнение сложности умножения") 
plot!(sizes, karatsuba_times, label="Алгоритм Карацубы", linestyle=:dash)
\end{lstlisting}
