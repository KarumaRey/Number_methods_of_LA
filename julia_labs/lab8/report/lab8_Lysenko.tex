\section{Цель работы}\label{Sect::task}
\par
\renewcommand{\labelenumii}{\arabic{enumi}.\arabic{enumii}.}
Реализовать и исследовать эффективности модификаций методов Штрассена и Винограда-Штрассена быстрого умножения матриц.

\section{Задачи}\label{Sect::task}
\par
\renewcommand{\labelenumii}{\arabic{enumi}.\arabic{enumii}.}
\begin{enumerate} 
    \item На основе уже реализованного в лабораторной работе №7 метода Штрассена необходимо выполнить его многопоточную версию.
    \item Реализовать классическую версию метода Винограда-Штрассена.
    \item Реализовать многопоточную версию метода Винограда-Штрассена.
    \item Сравнить скорости умножения матриц всеми реализованными методами: в лабораторной работе №7 и лабораторной работе №8.
    \item Построить на одной координатной плоскости разными цветами графики зависимости времени Т (мс) умножения двух матриц размера N x N стандартным алгоритмом, алгоритмом Винограда, методом Штрассена, многопоточным методом Штрассена, методом Винограда-Штрассена, многопоточным методом Винограда-Штрассена от размера матрицы N. N изменяется от 2 до 1024. Выбор значения размера матрицы N необходимо осуществлять по формуле N=$2^n$, при этом изменяя n от 1 до 10 строго. Во время построения графиков на координатной плоскости на горизонтальной оси время, которое затрачено на выполнение алгоритма перемножения матриц строго в миллисекундах.
\end{enumerate}

function standard(A, B)
    n = size(A, 1)
    C = zeros(Float64, n, n)
    for i in 1:n
        for j in 1:n
            for k in 1:n
                C[i, j] += A[i, k] * B[k, j]
            end
        end
    end
    return C
end


function winograd(A, B)
    n = size(A, 1)
    C = zeros(Float64, n, n)
    m = n ÷ 2

    for i in 1:n
        for j in 1:n
            for k in 1:m
                C[i,j] += (A[i, 2*k-1] + B[2*k,j]) * (B[2*k-1,j] + A[i, 2*k]) - A[i, 2*k-1] * A[i, 2*k] - B[2*k,j] * B[2*k-1,j]

            end
            
        end
    end
    return C
end

function strassen_winograd(A, B)
    n = size(A, 1)
    if n <= 64
        return standard(A, B)
    end
    if n % 2 != 0
        new_size = n + 1
        A_new = zeros(Float64, new_size, new_size)
        B_new = zeros(Float64, new_size, new_size)
        A_new[1:n, 1:n] = A
        B_new[1:n, 1:n] = B
        C_new = strassen_winograd(A_new, B_new)
        return C_new[1:n, 1:n]
    end

    k = n ÷ 2

    A11 = A[1:k, 1:k]
    A12 = A[1:k, k+1:n]
    A21 = A[k+1:n, 1:k]
    A22 = A[k+1:n, k+1:n]

    B11 = B[1:k, 1:k]
    B12 = B[1:k, k+1:n]
    B21 = B[k+1:n, 1:k]
    B22 = B[k+1:n, k+1:n]

    S2 = S1 - A11
    S3 = A11 - A21
    S4 = A12 - S2

    S8 = S6 - B21

    P6 = strassen_winograd(S4, B22)
    P7 = strassen_winograd(A22, S8)

    T2 = T1 + P4

    C11 = P2 + P3
    C12 = T1 + P5 + P6

    C22 = T2 + P5

    C = zeros(Float64, n, n)

function multi_treaded_strassen(A, B)
    n = size(A, 1)
    if n <= 64
        return standard(A, B)
    end
    if n % 2 != 0
        new_size = n + 1
        A_new = zeros(Float64, new_size, new_size)
        B_new = zeros(Float64, new_size, new_size)
        A_new[1:n, 1:n] = A
        B_new[1:n, 1:n] = B
        C_new = multi_treaded_strassen(A_new, B_new)
        return C_new[1:n, 1:n]
    end

    k = n ÷ 2
    A11, A12 = A[1:k, 1:k], A[1:k, k+1:end]
    A21, A22 = A[k+1:end, 1:k], A[k+1:end, k+1:end]
    B11, B12 = B[1:k, 1:k], B[1:k, k+1:end]
    B21, B22 = B[k+1:end, 1:k], B[k+1:end, k+1:end]

    P1, P2, P3, P4, P5, P6, P7 = Tuple{Matrix{Float64}, Matrix{Float64}, Matrix{Float64}, Matrix{Float64}, Matrix{Float64}, Matrix{Float64}, Matrix{Float64}}(zeros(k, k) for _ in 1:7)

    Threads.@threads for i in 1:7
        if i == 1
            P1 .= multi_treaded_strassen(A11 + A22, B11 + B22)
        elseif i == 2
            P2 .= multi_treaded_strassen(A21 + A22, B11)
        elseif i == 3
            P3 .= multi_treaded_strassen(A11, B12 - B22)
        elseif i == 4
            P4 .= multi_treaded_strassen(A22, B21 - B11)
        elseif i == 5
            P5 .= multi_treaded_strassen(A11 + A12, B22)
        elseif i == 6
            P6 .= multi_treaded_strassen(A21 - A11, B11 + B12)
        elseif i == 7
            P7 .= multi_treaded_strassen(A12 - A22, B21 + B22)
        end
    end

    C11 = P1 + P4 - P5 + P7
    C12 = P3 + P5
    C21 = P2 + P4
    C22 = P1 - P2 + P3 + P6

    C = [C11 C12; C21 C22]

    return C[1:size(A, 1), 1:size(B, 2)]
end



function multi_treaded_strassen_winograd(A, B)
    n = size(A, 1)
    if n <= 64
        return standard(A, B)
    end
    if n % 2 != 0
        new_size = n + 1
        A_new = zeros(Float64, new_size, new_size)
        B_new = zeros(Float64, new_size, new_size)
        A_new[1:n, 1:n] = A
        B_new[1:n, 1:n] = B
        C_new = multi_treaded_strassen_winograd(A_new, B_new)
        return C_new[1:n, 1:n]
    end

    k = n ÷ 2

    A11 = A[1:k, 1:k]
    A12 = A[1:k, k+1:n]
    A21 = A[k+1:n, 1:k]
    A22 = A[k+1:n, k+1:n]

    B11 = B[1:k, 1:k]
    B12 = B[1:k, k+1:n]
    B21 = B[k+1:n, 1:k]
    B22 = B[k+1:n, k+1:n]

    S1 = A21 + A22
    S2 = S1 - A11
    S3 = A11 - A21
    S4 = A12 - S2
    S5 = B12 - B11
    S6 = B22 - S5
    S7 = B22 - B12
    S8 = S6 - B21

    P1, P2, P3, P4, P5, P6, P7 = Tuple{Matrix{Float64}, Matrix{Float64}, Matrix{Float64}, Matrix{Float64}, Matrix{Float64}, Matrix{Float64}, Matrix{Float64}}(zeros(k, k) for _ in 1:7)
    
    Threads.@threads for i in 1:7
        if i == 1
            P1 .= multi_treaded_strassen_winograd(S2, S6)
        elseif i == 2
            P2 .= multi_treaded_strassen_winograd(A11, B11)
        elseif i == 3
            P3 .= multi_treaded_strassen_winograd(A12, B21)
        elseif i == 4
            P4 .= multi_treaded_strassen_winograd(S3, S7)
        elseif i == 5
            P5 .= multi_treaded_strassen_winograd(S1, S5)
        elseif i == 6
            P6 .= multi_treaded_strassen_winograd(S4, B22)
        elseif i == 7
            P7 .= multi_treaded_strassen_winograd(A22, S8)
        end
    end

    T1 = P1 + P2
    T2 = T1 + P4

    C11 = P2 + P3
    C12 = T1 + P5 + P6
    C21 = T2 - P7
    C22 = T2 + P5

    C = zeros(Float64, n, n)
    C[1:k, 1:k] = C11
    C[1:k, k+1:n] = C12
    C[k+1:n, 1:k] = C21
    C[k+1:n, k+1:n] = C22

    return C
end
